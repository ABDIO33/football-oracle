## 🔴 CRITICAL FIX #2: Class Weights في LabelSmoothingLoss
### الملف: train_v62.py | السطر: ~60
### التأثير: +2-3% | الوقت: 30 دقيقة

#### المشكلة:
- توزيع النتائج يتبع Power Law (1-0 = 20%، 4-4 = 0.01%)
- لا يوجد class_weight في أي دالة خسارة
- نتائج نادرة (3-2, 4-0, 4-3) يتوقعها الموديل عشوائياً

# إضافة بعد الاستيرادات
def compute_class_weights(y, method='sqrt_inv'):
    """حساب class weights بناءً على توزيع الفئات"""
    from collections import Counter
    counts = Counter(y)
    n_classes = 25
    weights = np.zeros(n_classes)
    total = len(y)
    
    for i in range(n_classes):
        freq = counts.get(i, 0) / total
        if method == 'inverse':
            weights[i] = 1.0 / max(freq, 0.001)
        elif method == 'sqrt_inv':
            weights[i] = 1.0 / np.sqrt(max(freq, 0.001))
        elif method == 'log_inv':
            weights[i] = 1.0 / np.log(max(freq, 0.001) + 1.1)
    
    # تطبيع إلى متوسط 1.0
    weights = weights / weights.mean()
    return torch.tensor(weights, dtype=torch.float32)


# ─── دالة خسارة جديدة مع Class Weights ───
class WeightedLabelSmoothingLoss(nn.Module):
    """Label smoothing + focal + class weights للتوزيع غير المتوازن"""
    def __init__(self, class_weights=None, smoothing=0.1, gamma=2.0):
        super().__init__()
        self.smoothing = smoothing
        self.gamma = gamma
        self.class_weights = class_weights  # Tensor [25]
    
    def forward(self, inputs, targets):
        n_classes = inputs.size(1)
        with torch.no_grad():
            smoothed = torch.full_like(inputs, self.smoothing / (n_classes - 1))
            smoothed.scatter_(1, targets.unsqueeze(1), 1.0 - self.smoothing)
        
        log_softmax = nn.functional.log_softmax(inputs, dim=1)
        loss = -(smoothed * log_softmax).sum(dim=1)
        
        # Focal modulation
        if self.gamma > 0:
            prob = torch.softmax(inputs, dim=1)
            p_t = prob.gather(1, targets.unsqueeze(1)).squeeze()
            loss = loss * ((1 - p_t) ** self.gamma)
        
        # Class weighting
        if self.class_weights is not None:
            weights = self.class_weights[targets]
            loss = loss * weights
        
        return loss.mean()


# في دالة التدريب الرئيسية:
criterion = WeightedLabelSmoothingLoss(

        expected_home = sum((s['home_goals'] * s['prob']/100) for s in scores)
        expected_away = sum((s['away_goals'] * s['prob']/100) for s in scores)
        
        confidence = scores[0]['prob'] - scores[3]['prob'] if len(scores) > 3 else scores[0]['prob'] - scores[1]['prob']

        return {

            'best_score': scores[0]['score'],

            'away_win_prob': away_win,
            'expected_home_goals': round(expected_home, 2),
            'expected_away_goals': round(expected_away, 2),
            'confidence': confidence,
            'model': 'V62Ensemble',
            'features_count': len(features_dict),
        }
    
    except Exception as e:
        log_error(f"ML prediction failed: {e}")

def analyze_match_deep(home_team, away_team, neutral_venue=False):
    """MASTER PREDICTOR: ML primary → Dixon-Coles backup"""
    
    start = time.time()
    result = {
        'match': f"{home_team} vs {away_team}",
        'neutral': neutral_venue,
        'home_team': home_team,
        'away_team': away_team,
    }
    
    # STEP 1: Try ML (V62Ensemble with 194 features)
    ml_result = predict_match_ml(home_team, away_team, neutral_venue)
    
    if ml_result and ml_result.get('confidence', 0) >= CONFIDENCE_THRESHOLD:
        result.update(ml_result)
        result['predictor'] = 'V62Ensemble (ML)'
        result['fallback'] = None
    else:
        # STEP 2: Fallback to Dixon-Coles
        dc_result = predict_match_dc(home_team, away_team, neutral_venue)
        result.update(dc_result)
        result['predictor'] = 'Dixon-Coles (Poisson)'
        result['fallback'] = 'Dixon-Coles backup'
        result['ml_reason'] = 'confidence too low' if ml_result else 'model not available'
    
    # STEP 3: Calibrate 1X2
    if _CALIBRATORS and len(_CALIBRATORS) == 3:
        h, d, a = _apply_calibration(
            result['home_win_prob'], result['draw_prob'], result['away_win_prob']
        )
        result['home_win_prob'] = h
        result['draw_prob'] = d
        result['away_win_prob'] = a
        result['calibration_applied'] = True
    
    # STEP 4: Market blend
    odds = get_live_odds(home_team, away_team)
    if odds:
        result = blend_with_market(result, odds)
        result['market_blend'] = True
    
    # STEP 5: Value bet detection
    if odds:
        value = detect_value_bet(result, odds)
        result['value_bet'] = value
    
    result['inference_time_ms'] = int((time.time() - start) * 1000)
    return result

def temporal_purged_cv(X, y, dates, n_splits=10, gap=5000):
    """
    Purged Walk-Forward Cross-Validation
    
    n_splits = 10 folds
    gap = 5000 samples (~6 months) between train/test
    
    Returns:
        - OOF exact score accuracy لكل fold
        - متوسط + انحراف معياري
        - قائمة بأوزان ensemble لكل fold
    """
    from sklearn.model_selection import TimeSeriesSplit
    
    idx = np.argsort(dates)
    X_sorted = X[idx]
    y_sorted = y[idx]
    
    tscv = TimeSeriesSplit(n_splits=n_splits, gap=gap)
    
    fold_results = []
    all_ood_preds = []
    all_ood_true = []
    ensemble_weights = []
    
    for fold, (train_idx, test_idx) in enumerate(tscv.split(X_sorted)):
        log(f"\n{'='*50}")
        log(f"Fold {fold+1}/{n_splits}")
        log(f"Train: {len(train_idx)} samples → Test: {len(test_idx)} samples")
        log(f"Gap: {gap} samples")
        
        X_train, X_test = X_sorted[train_idx], X_sorted[test_idx]
        y_train, y_test = y_sorted[train_idx], y_sorted[test_idx]
        
        # توصية: استخدام validation set منفصل للـ weight search
        val_size = min(10000, len(X_test))
        X_val, y_val = X_test[:val_size], y_test[:val_size]
        X_test, y_test = X_test[val_size:], y_test[val_size:]
        
        # حساب class weights
        class_weights = compute_class_weights(y_train)
        
        # تدريب
        fold_models = train_all_architectures(
            X_train, y_train, X_val, y_val, class_weights
        )
        
        # البحث عن أفضل أوزان ensemble على VALIDATION فقط
        best_weights = find_ensemble_weights(fold_models, X_val, y_val)
        ensemble_weights.append(best_weights)
        
        # تقييم على TEST (unseen)
        ensemble_probs = predict_ensemble(fold_models, best_weights, X_test)
        pred_classes = np.argmax(ensemble_probs, axis=1)
        fold_acc = np.mean(pred_classes == y_test) * 100
        
        fold_results.append(fold_acc)
        all_ood_preds.extend(pred_classes.tolist())
        all_ood_true.extend(y_test.tolist())
        
        log(f"✅ Fold {fold+1} OOF Accuracy: {fold_acc:.2f}%")
    
    oof_acc = np.mean(fold_results)
    oof_std = np.std(fold_results)
    
    log(f"\n{'='*50}")
    log(f"📊 OOF Exact Score Accuracy: {oof_acc:.2f}% ± {oof_std:.2f}%")
    log(f"Folds: {fold_results}")
    
    return {
        'oof_accuracy': oof_acc,
        'oof_std': oof_std,
        'per_fold': fold_results,
        'ensemble_weights': ensemble_weights,
    }
